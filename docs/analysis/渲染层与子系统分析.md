## 2. 音频采集：真实实现（重要修正）

### 2.1 AudioWorklet 完整源码（未混淆）

`C7lI2ZxR.js` 是唯一**完全未混淆**的业务文件——它是 AudioWorklet 处理器，
在独立线程运行，构建工具未对其做变量名混淆。源码可直接阅读：

```js
const PROCESSOR_NAME = 'desktop-app-audio-capture'
const SAMPLES_PER_MESSAGE = 1024

class 目标应用AudioCaptureProcessor extends AudioWorkletProcessor {
  constructor() {
    super()
    this.recording = false
    this.pendingSamples = new Float32Array(SAMPLES_PER_MESSAGE)
    this.pendingSampleCount = 0
    this.dataMessage = { type: 'recording:data', samples: null }

    this.port.onmessage = (event) => {
      const message = event.data
      if (message?.type === 'recording:start') { this.startRecording(); return }
      if (message?.type === 'recording:stop') {
        this.recording = false
        this.flushPendingSamples()
        this.port.postMessage({ type: 'recording:stopped' })
        this.pendingSampleCount = 0
      }
    }
  }

  flushPendingSamples() {
    if (this.pendingSampleCount === 0) return
    const samples = new Float32Array(this.pendingSampleCount)
    for (let i = 0; i < this.pendingSampleCount; i += 1) {
      samples[i] = this.pendingSamples[i]
    }
    this.pendingSampleCount = 0
    this.dataMessage.samples = samples
    // 用 transferable 移交 buffer，零拷贝
    this.port.postMessage(this.dataMessage, [samples.buffer])
    this.dataMessage.samples = null
  }

  appendInputChannels(channels) {
    if (channels.length === 1) { this.appendSamples(channels[0]); return }
    // 多声道：取算术平均做混音
    const frameCount = channels[0]?.length || 0
    let readOffset = 0
    while (readOffset < frameCount) {
      const availableSpace = SAMPLES_PER_MESSAGE - this.pendingSampleCount
      const samplesToCopy = Math.min(availableSpace, frameCount - readOffset)
      for (let i = 0; i < samplesToCopy; i += 1) {
        let sampleSum = 0
        for (const channel of channels) sampleSum += channel[readOffset + i]
        this.pendingSamples[this.pendingSampleCount + i] = sampleSum / channels.length
      }
      this.pendingSampleCount += samplesToCopy
      readOffset += samplesToCopy
      if (this.pendingSampleCount === SAMPLES_PER_MESSAGE) this.flushPendingSamples()
    }
  }

  process(inputs) {
    if (!this.recording) return true
    const input = inputs[0]
    if (!input || input.length === 0 || !input[0] || input[0].length === 0) return true
    this.appendInputChannels(input)
    return true
  }
}

registerProcessor(PROCESSOR_NAME, 目标应用AudioCaptureProcessor)
```

**关键设计点**：
- **1024 样本/消息**（16kHz 下约 64ms），比 ScriptProcessor 的 4096 帧更细粒度
- 用 **transferable `[samples.buffer]`** 移交内存所有权，主线程零拷贝接收
- 多声道用**算术平均**混音（非取首声道）
- `process()` 在未录音时立即 `return true`，不产生任何开销

### 2.2 AudioContext 配置

```js
// UfR9-a2Z.js
ud = '' + new URL('C7lI2ZxR.js', import.meta.url).href   // worklet 模块 URL
Ut = 0x3e80                                                // = 16000 Hz

class AudioCaptureManager {
  constructor(opts) {
    this.audioContext = new AudioContext({
      ...opts,
      sampleRate: opts?.sampleRate || Ut
    })
    // 关键：把 sink 设为 none，避免录音回放造成啸叫
    if (typeof this.audioContext.setSinkId === 'function') {
      this.audioContext.setSinkId({ type: 'none' }).catch(() => {})
    }
  }

  ensureAudioCaptureWorkletModuleLoaded() {
    return this.audioCaptureWorkletModulePromise ||= 
      this.audioContext.audioWorklet.addModule(ud).catch((e) => {
        this.audioCaptureWorkletModulePromise = undefined   // 失败可重试
        throw e
      })
  }

  async getAudioCaptureWorkletNode() {
    await this.ensureAudioCaptureWorkletModuleLoaded()
    return this.audioCaptureWorkletNode ||= new AudioWorkletNode(
      this.audioContext, 'desktop-app-audio-capture',
      { numberOfInputs: 1, numberOfOutputs: 0 }    // 纯输入节点，无输出
    )
  }
}
```

**`setSinkId({type:'none'})` 是重要细节**：把 AudioContext 输出路由到空设备，
否则内建麦克风会拾取扬声器声音形成啸叫。`numberOfOutputs: 0` 同理。

### 2.4 音量检测（独立于采集链路）

```js
// 用 AnalyserNode 而非从 PCM 算 RMS
function attachAnalyser(stream, ctx) {
  const source = ctx.createMediaStreamSource(stream)
  const analyser = ctx.createAnalyser()
  analyser.fftSize = 0x400           // 1024
  analyser.smoothingTimeConstant = 0.8
  source.connect(analyser)
  return { source, analyser }
}

async function openDeviceForTest(device, ctx) {
  const stream = await navigator.mediaDevices.getUserMedia({
    audio: { deviceId: device.deviceId }
  })
  const { source, analyser } = attachAnalyser(stream, ctx)
  return { stream, source, analyser, device, rmsSum: 0 }
}
```

**注意**：麦克风测试用的 `getUserMedia` 只传 `deviceId`，**不带任何约束参数**
（无 echoCancellation/noiseSuppression/autoGainControl）。这是测试原始音质，
与正式采集（可能带约束）分离。

音量计算方式（从字节数据算 RMS）：
```js
// 输入是 Uint8Array（AnalyserNode 的默认输出格式），中心值 128
let sum = 0
for (let i = 0; i < count; i++) {
  const centered = data[i] - 0x80
  sum += centered * centered
}
return Math.sqrt(sum / count) / 0x80
```

## 4. 遗留加密：硬编码 AES-256-CBC（安全发现）

主进程存在**第二套加密**，独立于 RSA 混合加密：

```js
const LEGACY_CIPHER = 'aes-256-cbc'
const LEGACY_KEY = Buffer.from(
  '7d4a8f2e6b9c3a1f5e8d2c7b4a9f6e3d1b5a2f9e6d3c0b7a4f1e8d5c2b9f6a3d', 'hex')
const IV_LENGTH = 0x10   // 16 字节

function decryptLegacy(value) {
  try {
    const parts = value.split(':')            // 格式 "iv:ciphertext"
    if (parts.length !== 2) throw new Error('无效的加密数据格式')
    const iv = Buffer.from(parts[0], 'base64')
    const ciphertext = parts[1]
    if (iv.length !== IV_LENGTH) throw new Error('无效的IV长度')
    const decipher = createDecipheriv(LEGACY_CIPHER, LEGACY_KEY, iv)
    let plain = decipher.update(ciphertext, 'base64', 'utf8')
    return plain += decipher.final('utf8')
  } catch {
    throw new Error('解密失败')
  }
}

// 调用点
parseAudioContext(raw) {
  if (!raw) return null
  const parsed = this.parseJsonSafely(raw)
  if (parsed !== undefined) return { value: parsed }
  // 解析失败 → 尝试遗留 AES 解密（旧版本数据格式）
  const decrypted = this.decryptLegacyAESAudioContext(raw)
  const reparsed = this.parseJsonSafely(decrypted)
  return reparsed === undefined ? null : { value: reparsed }
}

decryptLegacyAESAudioContext(v) {
  try { return decryptLegacy(v) } catch { return null }
}
```

**安全评估**：
- 密钥**硬编码在客户端**，任何持有应用副本的人都能提取
- 格式 `iv:ciphertext`，IV 由加密方提供（非固定），这一点比 RSA 路径的全零 IV 好
- 用途仅限**读取旧版本历史数据**（向后兼容），新数据走 RSA 信封
- 风险：用户升级前用旧版本产生的 `audio_context` 内容可被任意第三方解密。
  若旧版本曾用它加密敏感上下文，则该数据已实质泄露

**这是 AES-CBC 的经典误用场景**：固定密钥 + 客户端存储 = 无实际保密性。

### 5.2 联网检索引用结构（新发现）

```js
function extractGrounding(payload) {
  return {
    chunks: payload?.grounding_chunks || [],
    supports: payload?.grounding_supports || []
  }
}

function normalizeChunks(payload) {
  const { chunks } = extractGrounding(payload)
  return chunks?.length ? chunks.map((c, index) => ({
    index,
    domain: c.web?.domain || '',
    title: c.web?.title || '',
    uri: c.web?.uri || '',
    iconUrl: c.web?.icon_url || ''
  })) : []
}
```

**这是 AI 回答的「来源引用」功能**：`grounding_chunks` 携带网页来源（域名/标题/URL/图标），

`vertexaisearch.cloud.google.com/grounding-api-redirect/`（Google Vertex AI 的检索增强）。

```js
activateAppOnMacOS(bundleId) {
  const script = `
try
  tell application id "${bundleId}"
    if it is running then
      activate
    else
      launch
    end if
  end tell
on error
  do shell script "open -b ${bundleId}"
end try`
  exec(`osascript -e '${script}'`, () => {})
}
```

**双层兜底**：AppleScript 失败时用 `open -b`（按 bundle id 启动）。

### 6.3 Windows：PowerShell + WScript.Shell

```js
activateAppOnWindows(appName) {
  const name = appName.replace(/\.exe$/i, '')
  const script = `
 $name = "${name}";
 $proc = Get-Process -Name $name -ErrorAction SilentlyContinue |
         Where-Object { $_.MainWindowHandle -ne 0 } | Select-Object -First 1;
 if ($proc) {
   $wshell = New-Object -ComObject WScript.Shell;
   $success = $wshell.AppActivate($proc.Id);
   if ($success) { exit 0 } else { exit 1 }
 } else {
   exit 2
 }`
  // exit 0=成功 1=激活失败 2=进程不存在
}
```

**关键**：`Where-Object { $_.MainWindowHandle -ne 0 }` 过滤掉无窗口的后台进程——
同名进程可能有多个，只激活有主窗口的那个。

### 6.4 Linux：wmctrl + gtk-launch 兜底

```js
activateAppOnLinux(name) {
  exec(`wmctrl -a "${name}"`, (err) => {
    if (err) exec(`gtk-launch "${name}" || "${name}"`, () => {})
  })
}
```

## 10. 权限检查子进程（机制还原）

```js
async childProcessCheckAccessibility() {
  const binPath = join(libRoot, 'child-process-helper', 'bin', 'macos',
                       'macosCheckAccessibility')
  const start = performance.now()
  try {
    const { stdout } = await execFile(binPath)
    const elapsed = (performance.now() - start).toFixed(2)
    return stdout.trim() === 'true'          // 子进程输出字符串
  } catch {
    const elapsed = (performance.now() - start).toFixed(2)
    // 子进程失败 → 回退进程内检查
    return AccessibilityService.getInstance().checkAccessibilityPermission()
  }
}
```

独立二进制的符号表确认：`macosCheckAccessibility_arm64_temp.trusted: Bool`，
输出 `"true"` / `"false"`。

**为什么需要独立进程**：macOS 的 TCC（隐私数据库）对辅助功能权限有**进程级缓存**。
应用自身进程内的 `AXIsProcessTrusted()` 结果可能滞后（用户刚授权但进程未刷新）。
独立进程每次启动都重新查询 TCC，能拿到实时状态。

### 11.2 Mixpanel 上报链路

```js
// 渲染层 → 主进程
window.ipcRenderer.invoke('mixpanel:track-event', { eventKey, data })

// 主进程 → 服务端（经 RSA 加密）
async function sendMixpanelEvent(eventKey, data) {
  const user = await AuthService.getInstance().getCurrentUser()
  const payload = {
    event_key: eventKey,
    client_user_id: user.client_user_id || user.user_id,
    data: {
      environment: BUNDLE_ID === 'now.desktop.app' ? 'production' : 'development',
      event_source: IS_MAC ? 'desktop_macos' : 'desktop_windows',
      desktop-app_app_version: APP_VERSION,
      locale: store.get('preferredLanguage') ?? '',
      ...data
    }
  }
  const encrypted = await encryptValue(payload, {
    fallback: true,                              // ← 同样可降级明文
    context: 'mixpanel_electronMainProcessSendMixPanelEvent'
  })
  await api.post('/app/metrics_mp', { body: JSON.stringify({ info: encrypted }) })
}
```

**注意**：埋点数据**也走加密**，且同样 `fallback: true` ——加密失败时埋点明文上报，
其中含 `client_user_id`（用户标识）。
